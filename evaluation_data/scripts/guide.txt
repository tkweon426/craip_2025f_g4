# Install dependencies
conda activate language_command_handler
pip install rosbag-topic-remove
pip install evo

# Start your simulation
ros2 launch go1_simulation go1.launch.gazebo.launch.py use_gt_pose:=true
ros2 run unitree_guide2 junior_ctrl 
ros2 run go1_simulation move_go1_with_keyboard.py 
# And move your robot as you want

# Save data including gt pose
cd ~/Downloads
mkdir rosbag

ros2 bag record -o trajectory_gt1 \
/clock \
/cmd_vel \
/scan \
/camera_face/camera_info \
/camera_face/depth \
/camera_face/image \
/camera_face/points \
/camera_top/camera_info \
/camera_top/depth \
/camera_top/image \
/camera_top/points \
/imu_plugin/out \
/go1_pose \
/go1_pose_2d

# Check
cd ~/Downloads/rosbag
ros2 bag info trajectory_gt1

# Play (Test whether all the topics are alive)
# To visualize /scan in RViz2, set "Fixed Frame" as lidar_link
ros2 bag play trajectory_gt1

# Filter /go1_pose and /go1_pose_2d topic
conda activate language_command_handler
python filter_bag.py --src trajectory_gt1 --dst trajectory1

# Check new trajectory
ros2 bag info trajectory1

# Save /go1_pose as txt file for the evaluation
# Poses are saved as TUM format:
# timestamp x y z q_x q_y q_z q_w
# For the conversion, the code use only x, y, and yaw value from the /go1_pose and set z as 0
python save_go1_pose_as_txt.py trajectory_gt1 ./trajectory1/go1_pose_gt.txt
